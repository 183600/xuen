# ============================================================
# IP-448 Affective State Certificate (ASC) — 情感状态证书
# ============================================================
@dataclass
class AffectiveStateCertificate:
    """conformal 情感覆盖 + 情感控制屏障 + 信任校准 + 认知负荷 + 隐私"""
    humans: list['HumanProfile']          # 每个含 affect_label / observation_index
    alpha_aff: float = 0.05               # 情感覆盖风险
    phi_max: float = 0.7                  # 舒适代价上界
    eps_trust: float = 0.1                # 信任校准容差
    cl_max: float = 0.8                   # 认知负荷上界
    cbf_eta: float = 0.1                  # 离散 CBF 衰减系数
    eps_dp: float = 1.0                   # 差分隐私预算
    calibrator: 'ConformalCalibrator' = None
    affect_model: 'AffectDynamics' = None  # a_{t+1}=f_aff(a_t,u_t,c_t)+w

    def verify_conformal_affect(self, observations: Tensor) -> bool:
        """∀h_i: ℙ(a_i ∈ C_aff(x)) ≥ 1-α_aff（边际覆盖，经离线校准）"""
        if self.calibrator is None:
            return False
        pred_sets = self.calibrator.predict(observations)   # [N_obs, N_aff]
        for h in self.humans:
            if h.affect_label not in pred_sets[h.observation_index]:
                return False
        return True

    def verify_comfort_cbf(self, affect_traj: Tensor, actions: Tensor) -> bool:
        """h_aff(a_{t+1}) ≥ (1-η)h_aff(a_t) ∧ h_aff ≥ 0，取 conformal 集内最坏情况"""
        for t in range(len(affect_traj) - 1):
            for i, h in enumerate(self.humans):
                a_t = affect_traj[t, i]
                # 在 C_aff 内取使 h_aff 最小的最坏后继，保证鲁棒性
                a_next_wc = self.affect_model.worst_case(a_t, actions[t], i)
                h_t = self.phi_max - self._comfort_cost(a_t)
                h_next = self.phi_max - self._comfort_cost(a_next_wc)
                if h_next < 0:
                    return False
                if h_next < (1.0 - self.cbf_eta) * h_t:
                    return False
        return True

    def _comfort_cost(self, a: Tensor) -> float:
        """a = [valence, arousal, dominance, load, trust] ∈ [0,1]^5"""
        valence, arousal, dominance, load, trust = a.unbind()
        discomfort = 0.5 * (1.0 - valence) + 0.3 * arousal + 0.2 * load
        fear = torch.relu(arousal - 0.6) * (1.0 - dominance)
        overload = torch.relu(load - self.cl_max)
        # 过度信任（trust 远超真实可靠性）单独计入耦合项
        return float(0.5 * discomfort + 0.3 * fear + 0.2 * overload)

    def verify_trust_calibration(self, trust_est: Tensor,
                                 reliability: Tensor) -> bool:
        """|T̂_i - R_i| ≤ ε_trust，防止过度信任导致人类放弃监督"""
        return bool(torch.max(torch.abs(trust_est - reliability)) <= self.eps_trust)

    def verify_cognitive_load(self, loads: Tensor) -> bool:
        return bool(torch.max(loads) <= self.cl_max)

    def verify_privacy(self, mechanism: 'PrivacyMechanism') -> bool:
        """情感数据属敏感个人信息：ε-差分隐私与知情同意为硬前置条件"""
        return mechanism.epsilon <= self.eps_dp and mechanism.consent_obtained

    def verify_all(self, observations: Tensor, affect_traj: Tensor,
                   actions: Tensor, trust_est: Tensor, reliability: Tensor,
                   loads: Tensor, privacy: 'PrivacyMechanism') -> bool:
        return (self.verify_conformal_affect(observations) and
                self.verify_comfort_cbf(affect_traj, actions) and
                self.verify_trust_calibration(trust_est, reliability) and
                self.verify_cognitive_load(loads) and
                self.verify_privacy(privacy))


# ============================================================
# IP-449 Energy Optimization Certificate (EOC) — 能耗优化证书
# ============================================================
@dataclass
class EnergyOptimizationCertificate:
    """跨时域预算 + 退化/热约束 + 最优性间隙"""
    horizon: int
    E_budget: float
    E_reserve: float
    gamma: float = 0.01                   # 预算违反风险
    SoH_min: float = 0.8
    delta_SoH: float = 0.01
    T_max: float = 60.0
    eps_opt: float = 0.1                  # 最优性间隙
    power_model: 'PowerModel' = None
    degradation_model: 'DegradationModel' = None
    thermal_model: 'ThermalModel' = None

    def verify_budget(self, power_traj: Tensor, energy_std: Tensor) -> bool:
        """ℙ(E(t+H) ≤ E_budget - E_res) ≥ 1-γ，高斯最坏情况保守化"""
        if self.power_model is None:
            return False
        E = torch.cumsum(power_traj, dim=0)
        sigma = torch.sqrt(torch.cumsum(energy_std ** 2, dim=0))
        z = self._z_score(1.0 - self.gamma)
        return bool(torch.max(E + z * sigma) <= self.E_budget - self.E_reserve)

    def verify_degradation_thermal(self, power_traj: Tensor,
                                   temp_traj: Tensor) -> bool:
        if self.degradation_model is None or self.thermal_model is None:
            return False
        if self.degradation_model.delta_soh(power_traj) > self.delta_SoH:
            return False
        if self.degradation_model.soh_after(power_traj) < self.SoH_min:
            return False
        if float(torch.max(temp_traj)) > self.T_max:
            return False
        # 热降额一致性：温度越高，允许功率越低，防止「证书通过但电池热失控」
        return self.thermal_model.power_within_limit(power_traj, temp_traj)

    def verify_optimality_gap(self, J_pi: float, J_lb: float) -> bool:
        """J(π) ≤ (1+ε_opt)·J_LB，J_LB 由凸松弛/能量下界给出"""
        if J_lb <= 0:
            return False
        return J_pi <= (1.0 + self.eps_opt) * J_lb

    def verify_all(self, power_traj: Tensor, energy_std: Tensor,
                   temp_traj: Tensor, J_pi: float, J_lb: float) -> bool:
        return (self.verify_budget(power_traj, energy_std) and
                self.verify_degradation_thermal(power_traj, temp_traj) and
                self.verify_optimality_gap(J_pi, J_lb))

    @staticmethod
    def _z_score(p: float) -> float:
        # 正态分位数近似（p=0.99 时 ≈2.326）
        import math
        return math.sqrt(2.0) * math.erfinv(2.0 * p - 1.0)

# IP-450 情感-能量统一准入门（AEUC, v26）
# ============================================================
@dataclass
class AffectiveEnergyUnifiedCertificate:
    """九证一体：USMC + SMC + RMC + MCC + ASC + EOC + 情感-能量耦合"""
    usmc: 'UnifiedStructuralMechanicalCertificate'
    smc: 'SemanticMappingCertificate'
    rmc: 'ReconfigurableMechanismCertificate'
    mcc: 'MultiRobotCoordinationCertificate'
    asc: AffectiveStateCertificate
    eoc: EnergyOptimizationCertificate
    mu_min: float = 0.05                  # 耦合可行裕度下界
    eps_E: float = 0.02                   # 能量改进阈值（相对预算）
    eps_phi: float = 0.02                 # 舒适改进阈值（相对 Φ_max）
    pareto_samples: int = 128

    def admits(self, multi_plan: 'MultiRobotPlan', sensor_data: dict,
               gt_entities=None, queries=None, retrieved=None) -> bool:
        # 1. 前七证（v25）：USMC / SMC / RMC / MCC
        if not self._seven_certificates_ok(multi_plan, sensor_data,
                                           gt_entities, queries, retrieved):
            return False
        # 2. 情感状态证书（共享人类状态）
        if not self.asc.verify_all(
            sensor_data['affect_observations'],
            sensor_data['affect_traj'],
            sensor_data['human_actions'],
            sensor_data['trust_est'],
            sensor_data['reliability'],
            sensor_data['cognitive_load'],
            sensor_data['privacy'],
        ):
            return False
        # 3. 能耗优化证书
        if not self.eoc.verify_all(
            sensor_data['power_traj'],
            sensor_data['energy_std'],
            sensor_data['temp_traj'],
            sensor_data['J_pi'],
            sensor_data['J_lb'],
        ):
            return False
        # 4. 情感-能量耦合证书（Pareto + 裕度）
        if not self.verify_coupling(multi_plan, sensor_data):
            return False
        return True

    def verify_coupling(self, multi_plan: 'MultiRobotPlan',
                        sensor_data: dict) -> bool:
        """μ_couple ≥ μ_min 且运行点为 (ε_E, ε_Φ)-Pareto 非支配"""
        E = multi_plan.energy_estimate()
        phi = multi_plan.comfort_cost_estimate()
        mu_E = (self.eoc.E_budget - self.eoc.E_reserve - E) / self.eoc.E_budget
        mu_phi = (self.asc.phi_max - phi) / self.asc.phi_max
        if min(mu_E, mu_phi) < self.mu_min:
            return False
        # Pareto：邻近可行计划不得同时显著改进能耗与舒适
        for _ in range(self.pareto_samples):
            plan_p = multi_plan.perturb(scale=0.01)
            if not self._locally_feasible(plan_p):
                continue
            E_p = plan_p.energy_estimate()
            phi_p = plan_p.comfort_cost_estimate()
            if (E_p <= E - self.eps_E * self.eoc.E_budget and
                    phi_p <= phi - self.eps_phi * self.asc.phi_max):
                return False   # 存在双向改进 → 非 Pareto 有效
        return True

    def _locally_feasible(self, plan: 'MultiRobotPlan') -> bool:
        # 邻近计划仅需满足 EOC 预算与 ASC 舒适硬约束（其余证书由本地扰动保持）
        return (plan.energy_estimate() <= self.eoc.E_budget - self.eoc.E_reserve
                and plan.comfort_cost_estimate() <= self.asc.phi_max)

    def _seven_certificates_ok(self, multi_plan, sensor_data,
                               gt_entities, queries, retrieved) -> bool:
        # 复用 v25 的 USMC + SMC + RMC + MCC 判定（省略逐字段展开）
        return SRC_USC_ADMITS(self.usmc, self.smc, self.rmc, self.mcc,
                              multi_plan, sensor_data,
                              gt_entities, queries, retrieved)

## 3. 正确性论证（增量）

### 引理 13（情感状态证书的覆盖-屏障-校准三重保证）
**陈述**：`ASC.verify_all(...) = True` ⟹ 情感闭环同时满足：
(1) conformal 边际覆盖 ℙ(a∈C_aff(x)) ≥ 1-α_aff，
(2) 舒适代价沿轨迹不越界：h_aff(a_t) ≥ 0 且 h_aff(a_{t+1}) ≥ (1-η)h_aff(a_t)，
(3) 信任校准 |T̂-R| ≤ ε_trust 且 CL ≤ CL_max，且情感数据处理满足 ε_dp-差分隐私与知情同意。

**证明**：
1. `verify_conformal_affect` 直接复用离线校准的 split-conformal 预测器，由 conformal prediction 的边际覆盖定理（Vovk et al. 2005）给出 ℙ(a∈C_aff) ≥ 1-α_aff；覆盖是对联合分布的单次边际承诺，不是逐人条件承诺（见风险 R16）。
2. `verify_comfort_cbf` 对每个时刻在 C_aff 内取使 h_aff 最小的最坏后继，故若最坏后继仍满足离散 CBF 条件，则真实后继必满足；归纳得 h_aff(a_t) ≥ (1-η)^t h_aff(a_0) ≥ 0，舒适集前向不变。
3. 信任校准约束直接排除「人类因过度信任而退出监督」这一隐性失效模式；隐私与同意是情感数据合法可用的前置谓词，任一不满足即整证失败。
∎

### 引理 14（能耗优化证书的预算-退化-最优性三重保证）
**陈述**：`EOC.verify_all(...) = True` ⟹ 能耗计划同时满足：
(1) 机会约束预算 ℙ(E(t+H) ≤ E_budget-E_res) ≥ 1-γ，
(2) 退化/热约束 ΔSoH ≤ δ_SoH、SoH ≥ SoH_min、T ≤ T_max 且功率不超过热降额上限，
(3) 最优性间隙 J(π) ≤ (1+ε_opt)J_LB。

**证明**：
1. 高斯最坏情况保守化 E + z_{1-γ}σ ≤ E_budget-E_res 是机会约束的充分条件（Cantelli/Gaussian 界），故满足即保证违反概率 ≤ γ。
2. 退化模型给出功率轨迹到 ΔSoH/SoH 的确定性映射；热模型给出温度到允许功率的降额映射。二者同时满足排除「电学可行但热/寿命不可行」的计划。
3. J_LB 由功率模型的凸松弛（如忽略整数约束的 LP/二次规划下界）给出；间隙条件确保所选计划不是「安全但极低效」的退化解，避免证书被保守策略轻易满足而失去工程意义。
∎

### 引理 15（情感-能量耦合的裕度-Pareto 保序）
**陈述**：设 Φ*(E) 为在能量预算 E 下可达到的最小舒适代价。若 Φ* 关于 E 单调不增（能量越充裕，可实现的舒适代价越低），且认证计划 π 满足 μ_couple ≥ μ_min > 0 与 (ε_E, ε_Φ)-Pareto 非支配，则：
(a) π 位于情感-能量可行集 ℱ 的内部（存在对能耗与舒适同时恶化的鲁棒邻域），
(b) 任何预算收缩 ΔE ≤ μ_min·E_budget 仍使 π 保持可行，且不会因预算收紧而突然违反舒适约束。

**证明**：
(a) μ_E ≥ μ_min 与 μ_Φ ≥ μ_min 分别给出 E(π) ≤ (1-μ_min)E_budget 与 Φ(π) ≤ (1-μ_min)Φ_max，由 E、Φ 对计划参数的连续性，存在半径 δ>0 的邻域 N(π) 使 N(π) ⊆ ℱ，即 π 为内点。
(b) 由 Φ* 对 E 的单调性，预算收缩 ΔE 后新的最小舒适代价 Φ*(E-ΔE) ≥ Φ*(E)；而 π 的舒适代价 Φ(π) 只取决于动作参数、不随预算标签变化，故只要 π 的动作仍在能量可行域内（由 ΔE ≤ μ_min·E_budget 保证），舒适约束不会被预算收紧「挤破」。Pareto 非支配条件保证不存在被支配的伪最优运行点，避免证书把「又费能又不舒服」的计划当成前沿点。
∎

### 引理 16（九位一体单调退化链终极延伸）
**陈述**：任意单点/组合失效下，`Exec_v26 ⊆ Exec_v25 ⊆ ... ⊆ Exec_v18` 恒成立。

**证明**：在引理 12 基础上增加三条退化路径：
- **情感漂移/覆盖失效/舒适越界/信任失准** → `ASC.verify_all=false` → `AEUC` 拒证 → 退回 `Exec_v25`：关闭情感自适应交互，切换到预标定的「情感盲保守交互档」（固定低速、加大安全间距、显式提示人类接管），保留结构-语义-重构-协同七证。
- **能量预算超支/退化超限/热越界/最优性失真** → `EOC.verify_all=false` → `AEUC` 拒证 → 退回 `Exec_v25`：收缩任务范围、插入充电/换电窗口或跨机能量再分配，保留其余证书。
- **耦合裕度不足/Pareto 失配** → `verify_coupling=false` → `AEUC` 拒证 → 退回 `Exec_v25` 中预计算的「情感-能量锚点计划」（在保守预算与保守舒适档下已认证的 Pareto 点），禁止在未认证前沿上运行。
- **组合失效**：取交集，`Exec_v26` 进一步收缩，单调性由交集的单调性保持。
∎

### 定理（AEUC 九证准入门的安全-合规-可部署-可保护-结构-语义-重构-协同-情感-能量完备性）
**陈述**：对任意多机计划 Π 与人类集合 ℋ，`AEUC.admits(Π,ℋ) = True` ⟺ Π 同时满足：
(1)–(10) v25 的十项条件（物理接地、云边一致、少样本 conformal、专利保护、结构/力控/轻量、语义、重构、协同）；
(11) 情感覆盖、情感 CBF、信任校准、认知负荷与隐私五重达标；
(12) 能量机会约束预算、退化/热约束、最优性间隙三重达标，且情感-能量耦合裕度与 Pareto 非支配成立。

**证明**：`AEUC.admits` 是 (1)–(12) 的合取。必要性：任一条件缺失即返回 false，对应十类风险之一未被缓解。充分性：前七证由引理 4、8、12 给出单机到群体的安全闭环；ASC 由引理 13 给出人在回路的情感闭环；EOC 由引理 14 给出跨时域能量可行性；耦合由引理 15 给出内点与保序；四者合取得九位一体单调链。∎

## 6. 本轮最大风险（最终确认）

**一句话**：**情感推断的个体/文化 OOD 使 conformal 边际覆盖无法兑现逐人舒适保证，电池退化/热模型漂移使 EOC 的确定性映射在老化后失真，而非凸耦合 Pareto 又只能用采样近似——三者叠加时，AEUC 可能在「人因敏感 + 能量紧张 + 分布漂移」的真实工况下既错误放行慢性不安全计划，又因九证交集过保守而近乎拒绝一切计划，使「事前拦截」的承诺在长期人机共处中失效。**
